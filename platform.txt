# For more info:
# https://github.com/arduino/Arduino/wiki/Arduino-IDE-1.5---3rd-party-Hardware-specification

name=Arduino RISC-V Boards
version=1.0.0

# Compile variables
# ------------------------

compiler.warning_flags=-w
compiler.warning_flags.none=-w
compiler.warning_flags.default=
compiler.warning_flags.more=-Wall
compiler.warning_flags.all=-Wall -Wextra

compiler.optimization_flags=-Os
compiler.optimization_flags.release=-Os
compiler.optimization_flags.debug=-Og

compiler.c.extra_flags=-march=rv64imac -mabi=lp64d
compiler.cpp.extra_flags=-march=rv64imac -mabi=lp64d
compiler.S.extra_flags=-march=rv64imac -mabi=lp64d

compiler.c.elf.flags=  -nostdlib -nodefaultlibs -nostartfiles -Wl,-z,noexecstack,--gc-sections -Wl,--exclude-libs,ALL {build.extra_ldflags}
compiler.c.flags=-c -fPIC -mcmodel=medany {compiler.warning_flags} {compiler.optimization_flags} -g3 -nostdlib {build.defines} -MMD -std=gnu11 -march=rv64imafdc_zicsr -mabi=lp64d -fmessage-length=0 -fsigned-char -ffunction-sections -fdata-sections -fmessage-length=0 -fno-builtin
compiler.cpp.flags=-c {compiler.warning_flags} {compiler.optimization_flags} -fPIC -mcmodel=medany -march=rv64imafdc -mabi=lp64d -g3 -fno-use-cxa-atexit -fno-rtti -fno-exceptions -MMD -nostdlib {build.defines} -std=gnu++17 -fsigned-char -ffunction-sections -fdata-sections -fmessage-length=0 -fno-builtin
compiler.S.flags=-c -g -x assembler-with-cpp {compiler.optimization_flags} -march=rv64imafdc_zicsr -mabi=lp64d -fsigned-char -ffunction-sections -fdata-sections
compiler.ar.flags=rcs

compiler.elf2hex.bin.flags=-O binary -j .text -j .data
compiler.elf2hex.hex.flags=-O ihex -j .text -j .data

compiler.path=/opt/riscv/bin/
compiler.c.cmd=riscv64-unknown-elf-gcc
compiler.c.elf.cmd=riscv64-unknown-elf-g++
compiler.S.cmd=riscv64-unknown-elf-gcc
compiler.cpp.cmd=riscv64-unknown-elf-g++
compiler.ar.cmd=riscv64-unknown-elf-ar
compiler.objcopy.cmd=riscv64-unknown-elf-objcopy

compiler.size.cmd=riscv64-unknown-elf-size
compiler.define=-DARDUINO=
build.extra_flags=-include syscalls.cpp
build.extra_ldflags=
compiler.S.extra_flags=

# Compile Patterns
# -----------------------

recipe.c.o.pattern="{compiler.path}{compiler.c.cmd}" {compiler.c.flags} {includes} -c "{source_file}" -o "{object_file}"
recipe.cpp.o.pattern="{compiler.path}{compiler.cpp.cmd}" {compiler.cpp.flags} {includes} -c "{source_file}" -o "{object_file}"

## Compile Assembly files
recipe.S.o.pattern="{compiler.path}{compiler.S.cmd}" {compiler.S.flags} -DARDUINO={runtime.ide.version} -DARDUINO_{build.board} -DARDUINO_ARCH_{build.arch} -DARDUINO_ARCH_RISCV {includes} {build.extra_flags} {compiler.S.extra_flags} "{source_file}" -o "{object_file}"

## Create archive
recipe.ar.pattern="{compiler.path}{compiler.ar.cmd}" {compiler.ar.flags} "{archive_file_path}" "{object_file}"

## Linker
recipe.c.combine.pattern="{compiler.path}{compiler.c.elf.cmd}" {compiler.c.elf.flags} {build.extra_flags} -o "{build.path}/{build.project_name}.elf" "-L{build.path}" "-T{runtime.platform.path}/variants/{build.variant}/linker_scripts/link.ld" {object_files} "{runtime.platform.path}/cores/arduino/syscalls.cpp"  -Wl,--whole-archive "{build.path}/core/core.a" -Wl,--no-whole-archive -Wl,-Map={build.path}/{build.project_name}.map
## Compute size
recipe.size.pattern="{compiler.path}riscv64-unknown-elf-size" -A "{build.path}/{build.project_name}.elf"
